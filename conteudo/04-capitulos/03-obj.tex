\chapter{OBJETIVOS}

\section{Objetivo Geral}
Analisar as contribuições das práticas pedagógicas críticas no ensino de Geografia para a formação da consciência espacial e cidadã dos estudantes.

\section{Objetivos Específicos}
\begin{itemize}
    \item Investigar os conceitos de espaço e produção do espaço à luz da geografia crítica de \cite{milton-ne};
    \item Relacionar as técnicas de ensino com as abordagens contemporâneas de avaliação da aprendizagem;
    \item Avaliar o potencial dessas práticas para a formação de sujeitos críticos e conscientes de seu papel no espaço urbano.
\end{itemize}
