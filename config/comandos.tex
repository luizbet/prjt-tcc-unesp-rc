% ============================================================
% FORMATAÇÃO DE CAPÍTULOS E SEÇÕES
% ============================================================

% 1. Remove qualquer formatação anterior de capítulos
\titleformat{\chapter}[block]{}{}{}{}
\titlespacing*{\chapter}{0pt}{0pt}{0pt}

% 2. Cria o comando para o título do capítulo (centralizado, 12pt, negrito, maiúsculo)
\newcommand{\meuTituloCapitulo}[1]{%
    \vspace{0.3cm} % Espaço antes do capítulo
    \begin{center}
        {\fontsize{12pt}{14pt}\selectfont\bfseries \MakeUppercase{#1}} % Fonte reduzida
    \end{center}
    \vspace{0.1cm} % Espaço depois do capítulo
}

% 3. Redefine o comando \chapter usando a classe book nativa
\makeatletter
\renewcommand{\chapter}[1]{%
    \par\refstepcounter{chapter}%
    \meuTituloCapitulo{#1}%
    \addcontentsline{toc}{chapter}{\protect\numberline{\thechapter}#1}%
}
\makeatother

% 4. Configuração das seções (esquerda, negrito)
\titleformat{\section}
    {\normalfont\normalsize\bfseries\raggedright}
    {\thesection}
    {0.5em}
    {}

\titleformat{\subsection}
    {\normalfont\normalsize\bfseries\raggedright}
    {\thesubsection}
    {0.5em}
    {}

\titleformat{\subsubsection}
    {\normalfont\normalsize\itshape\raggedright}
    {\thesubsubsection}
    {0.5em}
    {}

\newcommand{\figura}[4]{%
    \begin{figure}[H]
        \centering
        \includegraphics[width=#1]{#2}
        \caption{#3}
        \label{fig:#4}
    \end{figure}
}

\newcommand{\tabela}[4]{%
    \begin{table}[H]
        \centering
        \caption{#1}
        \label{tab:#2}
        #3
        \fonte{#4}
    \end{table}
}

\newcommand{\fonte}[1]{%
    \vspace{0.2cm}
    \par\small\textbf{Fonte:} #1
}

\newcommand{\nota}[1]{%

}
